・セグメント木(RMQ) //O(logN)


struct SegTree {
  int n;
  vi seg;
  SegTree(int sz) : n(sz), seg(2*sz,inf) {}
  void set(int i, int x) {
    i += n;
    seg[i] = x;
    while (i >>= 1) seg[i] = min(seg[i<<1],seg[i<<1|1]);
  }
  int prod(int l, int r) {
    l += n; r += n;
    int ans = inf;
    while (l < r) {
      if (l & 1) chmin(ans,seg[l++]);
      if (r & 1) chmin(ans,seg[--r]);
      l >>= 1; r >>= 1;
    }
    return ans;
  }
};

・遅延評価セグメント木(区間加算、区間最小値取得) //O(logN)で区間への加算


struct Lazy_SegTree {
  int n; vi seg,lazy;
  Lazy_SegTree(int sz) {
    n = 1;
    while (n < sz) n <<= 1;
    seg.resize(2*n,inf);
    lazy.resize(2*n,0);
  }
  void apply(int i, int x) {
    seg[i] += x; lazy[i] += x;
  }
  void push(int i) {
    if (!lazy[i]) return;
    if (i < n) {
      apply(i<<1,lazy[i]);
      apply(i<<1|1,lazy[i]);
    }
    lazy[i] = 0;
  }
  void add(int k, int l, int r, int x, int a, int b) {
    if (a >= r || b <= l) return;
    if (a <= l && r <= b) {apply(k,x); return;}
    push(k);
    add(k<<1,l,(l+r)/2,x,a,b);
    add(k<<1|1,(l+r)/2,r,x,a,b);
    seg[k] = min(seg[k<<1],seg[k<<1|1]);
  }
  void add(int l, int r, int x) {
    add(1,0,n,x,l,r);
  }
  int prod(int k, int l, int r, int a, int b) {
    if (a >= r || b <= l) return inf;
    if (a <= l && r <= b) return seg[k];
    push(k);
    return min(prod(k<<1,l,(l+r)/2,a,b),
      prod(k<<1|1,(l+r)/2,r,a,b));
  }
  int prod(int l, int r) {
    return prod(1,0,n,l,r);
  }
};

・BIT //O(logN)


struct BIT {
  int n;
  vi bit;
  BIT(int sz) : n(sz), bit(sz+1,0) {}
  void add(int x, int y) {
    x++;
    for (int i = x; i <= n; i += i&-i) bit[i] += y;
  }
  //[0,x)の和
  int sum(int x) {
    int res = 0;
    for (int i = x; i > 0; i -= i&-i) res += bit[i];
    return res;
  }
  //[l,r)の和
  int prod(int l, int r) {
    return sum(r) - sum(l);
  }
};

・遅延評価セグメント木(区間代入、区間最大値取得) //O(logN)で区間へ代入


struct Lazy_SegTree {
  int n; vi seg,lazy;
  Lazy_SegTree(int sz) {
    n = 1;
    while (n < sz) n <<= 1;
    seg.resize(2*n,0);
    lazy.resize(2*n,-1);
  }
  void apply(int i, int x) {
    seg[i] = x; lazy[i] = x;
  }
  void push(int i) {
    if (lazy[i] == -1) return;
    if (i < n) {
      apply(i<<1,lazy[i]);
      apply(i<<1|1,lazy[i]);
    }
    lazy[i] = -1;
  }
  void upd(int k, int l, int r, int x, int a, int b) {
    if (a >= r || b <= l) return;
    if (a <= l && r <= b) {apply(k,x); return;}
    push(k);
    upd(k<<1,l,(l+r)/2,x,a,b);
    upd(k<<1|1,(l+r)/2,r,x,a,b);
    seg[k] = max(seg[k<<1],seg[k<<1|1]);
  }
  void upd(int l, int r, int x) {
    upd(1,0,n,x,l,r);
  }
  int prod(int k, int l, int r, int a, int b) {
    if (a >= r || b <= l) return 0;
    if (a <= l && r <= b) return seg[k];
    push(k);
    return max(prod(k<<1,l,(l+r)/2,a,b),prod(k<<1|1,(l+r)/2,r,a,b));
  }
  int prod(int l,int r) {
    return prod(1,0,n,l,r);
  }
};
